#   apt-get install libapr1-dev    # C binary-trees (apr_pools.h)
#   apt-get install libpcre3-dev   # C regex-redux (pcre.h)
#   apt-get install libhts-dev     # C k-nucleotide (htslib/khash.h)
#   plus a C/C++ toolchain with OpenMP (gcc, g++, libgomp).
